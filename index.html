import { createClient } from "npm:@supabase/supabase-js@2"; const corsHeaders = { "Access-Control-Allow-Origin": "*", "Access-Control-Allow-Headers": "authorization, x-client-info, apikey, content-type", "Access-Control-Allow-Methods": "POST, OPTIONS", }; Deno.serve(async (req) => { // Handle browser preflight request if (req.method === "OPTIONS") { return new Response("ok", { headers: corsHeaders, }); } try { // Get the logged-in user's authorization token const authHeader = req.headers.get("Authorization"); if (!authHeader) { return new Response( JSON.stringify({ error: "You must be logged in to search properties.", }), { status: 401, headers: { ...corsHeaders, "Content-Type": "application/json", }, }, ); } // Create Supabase client const supabase = createClient( Deno.env.get("SUPABASE_URL")!, Deno.env.get("SUPABASE_ANON_KEY")!, { global: { headers: { Authorization: authHeader, }, }, }, ); // Verify the logged-in user const { data: { user }, error: userError, } = await supabase.auth.getUser(); if (userError || !user) { return new Response( JSON.stringify({ error: "Your login session is invalid or expired.", }), { status: 401, headers: { ...corsHeaders, "Content-Type": "application/json", }, }, ); } // Make sure the RentCast secret exists const rentCastKey = Deno.env.get("RENTCAST_API_KEY"); if (!rentCastKey) { return new Response( JSON.stringify({ error: "RentCast API key is not configured in Supabase.", }), { status: 500, headers: { ...corsHeaders, "Content-Type": "application/json", }, }, ); } // Read the search request const body = await req.json(); const city = String(body.city || "").trim(); const state = String(body.state || "").trim(); const zipCode = String(body.zipCode || "").trim(); const minPrice = body.minPrice ? String(body.minPrice) : ""; const maxPrice = body.maxPrice ? String(body.maxPrice) : ""; const bedrooms = body.bedrooms ? String(body.bedrooms) : ""; const propertyType = body.propertyType ? String(body.propertyType) : ""; const limit = Math.min( Math.max(Number(body.limit) || 10, 1), 50, ); // At least one location is required if (!city && !zipCode) { return new Response( JSON.stringify({ error: "Enter a city or ZIP code.", }), { status: 400, headers: { ...corsHeaders, "Content-Type": "application/json", }, }, ); } // Build RentCast request const params = new URLSearchParams(); if (city) { params.set("city", city); } if (state) { params.set("state", state); } if (zipCode) { params.set("zipCode", zipCode); } if (minPrice) { params.set("price", `${minPrice}:`); } if (maxPrice) { if (minPrice) { params.set("price", `${minPrice}:${maxPrice}`); } else { params.set("price", `:${maxPrice}`); } } if (bedrooms) { params.set("bedrooms", `${bedrooms}:`); } if (propertyType) { params.set("propertyType", propertyType); } params.set("status", "Active"); params.set("limit", String(limit)); const rentCastUrl = `https://api.rentcast.io/v1/listings/sale?${params.toString()}`; // Ask RentCast for the actual properties const rentCastResponse = await fetch(rentCastUrl, { method: "GET", headers: { Accept: "application/json", "X-Api-Key": rentCastKey, }, }); const rentCastData = await rentCastResponse.json(); if (!rentCastResponse.ok) { return new Response( JSON.stringify({ error: "RentCast request failed.", details: rentCastData, }), { status: rentCastResponse.status, headers: { ...corsHeaders, "Content-Type": "application/json", }, }, ); } // Return only the useful data to Propelora return new Response( JSON.stringify({ success: true, user_id: user.id, count: Array.isArray(rentCastData) ? rentCastData.length : 0, listings: rentCastData, }), { status: 200, headers: { ...corsHeaders, "Content-Type": "application/json", }, }, ); } catch (error) { console.error(error); return new Response( JSON.stringify({ error: "Something went wrong while searching properties.", details: error instanceof Error ? error.message : String(error), }), { status: 500, headers: { ...corsHeaders, "Content-Type": "application/json", }, }, ); } });
